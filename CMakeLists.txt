cmake_minimum_required(VERSION 3.24)
project(astroattitude VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
option(AA_BUILD_TESTS "Build tests and synthetic diagnostics" ON)
option(AA_BUILD_TOOLS "Build offline conversion helpers" ON)
option(AA_SANITIZE "Enable address/undefined sanitizers" OFF)
if(AA_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
foreach(dep erfa eigen json sha)
 if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/${dep}")
  message(FATAL_ERROR "Run python3 tools/fetch_dependencies.py first")
 endif()
endforeach()
file(GLOB ERFA_SOURCES CONFIGURE_DEPENDS .deps/erfa/src/*.c)
list(FILTER ERFA_SOURCES EXCLUDE REGEX "/t_erfa_c(_extra)?\\.c$")
add_library(erfa STATIC ${ERFA_SOURCES})
target_include_directories(erfa PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/.deps/erfa/src>)
target_compile_definitions(erfa PRIVATE PACKAGE_VERSION="2.0.1" PACKAGE_VERSION_MAJOR=2 PACKAGE_VERSION_MINOR=0 PACKAGE_VERSION_MICRO=1 SOFA_VERSION="20231011")
add_library(astroattitude STATIC src/resources.cpp src/astrometry.cpp src/solver.cpp src/extraction.cpp src/calibration.cpp src/alignment.cpp)
target_include_directories(astroattitude PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(astroattitude PRIVATE .deps/eigen .deps/json/include .deps/sha .deps/erfa/src)
target_compile_definitions(astroattitude PRIVATE EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
target_link_libraries(astroattitude PRIVATE erfa)
add_library(astroattitude_c SHARED src/c_api.cpp)
target_link_libraries(astroattitude_c PRIVATE astroattitude)
target_include_directories(astroattitude_c PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(astroattitude_c PRIVATE .deps/eigen .deps/json/include .deps/sha .deps/erfa/src)
target_compile_definitions(astroattitude_c PRIVATE EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
add_library(astroattitude_c_static STATIC src/c_api.cpp)
target_link_libraries(astroattitude_c_static PRIVATE astroattitude)
target_include_directories(astroattitude_c_static PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(astroattitude_c_static PRIVATE .deps/eigen .deps/json/include .deps/sha .deps/erfa/src)
target_compile_definitions(astroattitude_c_static PRIVATE EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
set_target_properties(astroattitude_c PROPERTIES VERSION 0.1.0 SOVERSION 1)
if(ANDROID)
 add_library(astroattitude_jni SHARED bindings/android/native/bridge.cpp src/c_api.cpp)
 target_link_libraries(astroattitude_jni PRIVATE astroattitude)
 target_include_directories(astroattitude_jni PRIVATE .deps/eigen .deps/json/include .deps/sha .deps/erfa/src)
 target_compile_definitions(astroattitude_jni PRIVATE EIGEN_MPL2_ONLY)
 target_link_options(astroattitude_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/bindings/android/native/exports.map")
endif()
if(AA_BUILD_TOOLS)
 add_executable(aa_propagate tools/propagate.cpp)
 target_link_libraries(aa_propagate PRIVATE erfa)
endif()
if(AA_BUILD_TESTS)
 enable_testing()

endif()
include(GNUInstallDirs)
install(TARGETS astroattitude erfa astroattitude_c astroattitude_c_static EXPORT astroattitudeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/astroattitude)
install(DIRECTORY notices/ DESTINATION ${CMAKE_INSTALL_DATADIR}/astroattitude/notices)

 target_sources(aa_benchmark PRIVATE tests/allocation_probe.cpp)

include(CMakePackageConfigHelpers)
set_target_properties(astroattitude PROPERTIES EXPORT_NAME core)
set_target_properties(astroattitude_c PROPERTIES EXPORT_NAME c)
set_target_properties(astroattitude_c_static PROPERTIES EXPORT_NAME c_static)

write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/astroattitudeConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT astroattitudeTargets NAMESPACE astroattitude:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/astroattitude)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/astroattitudeConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/astroattitudeConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/astroattitude)
option(AA_BUILD_EXAMPLES "Build standalone synthetic usage examples" OFF)
