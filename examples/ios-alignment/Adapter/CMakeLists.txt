cmake_minimum_required(VERSION 3.24)
project(AlignmentAdapter LANGUAGES C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_PYTHON_WRAPPER OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT "../../.." ABSOLUTE)
add_subdirectory("${ROOT}/.deps/apriltag" apriltag)
add_library(AlignmentAdapter STATIC AlignmentAdapter.c)
target_include_directories(AlignmentAdapter PRIVATE "${ROOT}/.deps/apriltag" "${ROOT}/.deps/erfa/src")
